rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Admin-E-Mail
    function isAdmin() {
      return request.auth != null
             && request.auth.token.email == 'info@segelfliegen-altdorf.de';
    }

    // Nicht-anonymer User (allgemein eingeloggt)
    function isLoggedIn() {
      return request.auth != null
             && request.auth.token.firebase.sign_in_provider != 'anonymous';
    }

    // Gutschein-User oder Admin
    function isGutscheinUser() {
      return isAdmin()
             || (request.auth != null && request.auth.token.email == 'gutschein@segelfliegen-altdorf.de');
    }

    // Bestellungs-User oder Admin
    function isBestellungUser() {
      return isAdmin()
             || (request.auth != null && request.auth.token.email == 'bestellung@segelfliegen-altdorf.de');
    }

    // News, Preise, Flugzeugpark — öffentlich lesbar, nur Admin schreibt
    match /news/{doc} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /prices/{doc} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /aircraft/{doc} {
      allow read: if true;
      allow write: if isAdmin();
    }

    match /events/{doc} {
      allow read: if true;
      allow write: if isAdmin();
    }

    // Gutscheine — Admin + Gutschein-User
    match /vouchers/{doc} {
      allow read: if isGutscheinUser();
      allow write: if isGutscheinUser();
    }

    // Bestellungen — anonyme User dürfen erstellen, Admin + Bestellungs-User lesen/bearbeiten
    // Schema- und Größenlimits beim Create verhindern DoS und Spam-Submissions
    match /voucherOrders/{doc} {
      allow create: if request.auth != null
                    && request.resource.data.keys().hasOnly([
                         'name', 'email', 'telefon', 'flugart', 'zusatzzeit',
                         'wert', 'empfaenger', 'grusstext', 'zustellung',
                         'wertAnzeigen', 'status', 'paid', 'timestamp',
                         'flugdauer', 'anlass', 'schreibstil'
                       ])
                    && request.resource.data.name is string
                    && request.resource.data.name.size() <= 200
                    && request.resource.data.email is string
                    && request.resource.data.email.size() <= 200
                    && request.resource.data.telefon is string
                    && request.resource.data.telefon.size() <= 50
                    && request.resource.data.flugart is string
                    && request.resource.data.flugart.size() <= 100
                    && request.resource.data.empfaenger is string
                    && request.resource.data.empfaenger.size() <= 200
                    && (!('grusstext' in request.resource.data)
                        || (request.resource.data.grusstext is string
                            && request.resource.data.grusstext.size() <= 2000))
                    && (!('zustellung' in request.resource.data)
                        || (request.resource.data.zustellung is string
                            && request.resource.data.zustellung.size() <= 200))
                    && (!('wert' in request.resource.data)
                        || (request.resource.data.wert is string
                            && request.resource.data.wert.size() <= 20))
                    && (!('zusatzzeit' in request.resource.data)
                        || (request.resource.data.zusatzzeit is string
                            && request.resource.data.zusatzzeit.size() <= 10))
                    && (!('flugdauer' in request.resource.data)
                        || (request.resource.data.flugdauer is string
                            && request.resource.data.flugdauer.size() <= 200))
                    && (!('anlass' in request.resource.data)
                        || (request.resource.data.anlass is string
                            && request.resource.data.anlass.size() <= 200))
                    && (!('schreibstil' in request.resource.data)
                        || (request.resource.data.schreibstil is string
                            && request.resource.data.schreibstil.size() <= 200))
                    // Zahlungsstatus & Metadaten dürfen beim Anlegen nicht gefälscht werden
                    // (verhindert, dass ein anonymer Nutzer paid:true / anderen Status setzt):
                    && (!('paid' in request.resource.data)
                        || request.resource.data.paid == false)
                    && (!('status' in request.resource.data)
                        || request.resource.data.status == 'neu')
                    && (!('timestamp' in request.resource.data)
                        || request.resource.data.timestamp is number)
                    && (!('wertAnzeigen' in request.resource.data)
                        || request.resource.data.wertAnzeigen is bool);
      allow read, update, delete: if isBestellungUser();
    }

    // Widerruf-Protokoll (§ 356a BGB) — nur serverseitig via Admin-SDK geschrieben
    // (Admin-SDK umgeht diese Rules). Kein allow write → kein Client kann schreiben.
    match /widerrufe/{doc} {
      allow read: if isAdmin();
    }
  }
}
